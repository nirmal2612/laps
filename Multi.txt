using System;
using System.Runtime.InteropServices;

public class ConsoleHelper {
    const uint ENABLE_QUICK_EDIT = 0x0040;
    const int STD_INPUT_HANDLE = -10;

    [DllImport("kernel32.dll", SetLastError = true)]
    static extern IntPtr GetStdHandle(int nStdHandle);

    [DllImport("kernel32.dll")]

    [DllImport("kernel32.dll")]

        IntPtr consoleHandle = GetStdHandle(STD_INPUT_HANDLE);
        uint consoleMode;
        if (GetConsoleMode(consoleHandle, out consoleMode)) {

            SetConsoleMode(consoleHandle, consoleMode);

}

# Load authorized users from file
function LoadAuthorizedUsersFromFile {
    try {
        $userFile = Join-Path $SyncState.UserDetailsPath "authorized_users.json"
        if (Test-Path $userFile) {
            $content = Get-Content $userFile -Raw -ErrorAction SilentlyContinue
            if ($content) {
                $loaded = $content | ConvertFrom-Json -ErrorAction SilentlyContinue
                if ($loaded) {
                    [System.Threading.Monitor]::Enter($SyncState.AuthorizedUsers)
                    try {
                        $SyncState.AuthorizedUsers.Clear()
                        $loaded.PSObject.Properties | ForEach-Object {
                            $SyncState.AuthorizedUsers[$_.Name] = @{
                                username   = $_.Value.username
                                name       = $_.Value.name
                                authorized = $_.Value.authorized
                                addedBy    = $_.Value.addedBy
                                addedDate  = $_.Value.addedDate
                            }
                        }
                    } finally {
                        [System.Threading.Monitor]::Exit($SyncState.AuthorizedUsers)
                    }
                }
            }
        }
    } catch {
        Write-Host "Warning: Could not load authorized users: $_" -ForegroundColor Yellow
    }
}

$WorkerScriptBlock = {
    param(
        [System.Net.HttpListenerContext]$Context,

    # Re-import LAPS module inside worker thread
    Import-Module LAPS -ErrorAction SilentlyContinue

    $Request  = $Context.Request

    # --- HELPER FUNCTIONS WITHIN WORKER RUNSPACE ---
    function SafeHtmlEncode ([string]$Text) {
        if ([string]::IsNullOrEmpty($Text)) { return $Text }
        return $Text -replace '&', '&amp;' -replace '<', '&lt;' -replace '>', '&gt;' -replace '"', '&quot;' -replace "'", '&#39;'
    }

    function ParseFormData ([string]$Body) {
        $result = @{}
        if ([string]::IsNullOrEmpty($Body)) { return $result }
        $pairs = $Body -split '&'
        foreach ($pair in $pairs) {
            if ([string]::IsNullOrEmpty($pair)) { continue }
            $kv = $pair -split '=', 2
            if ($kv.Count -eq 2) {
                $result[[System.Web.HttpUtility]::UrlDecode($kv[0])] = [System.Web.HttpUtility]::UrlDecode($kv[1])
            }
        }
        return $result
    }

    function ParseCookies ([string]$CookieHeader) {
        $result = @{}
        if ([string]::IsNullOrEmpty($CookieHeader)) { return $result }
        $cookies = $CookieHeader -split '; '
        foreach ($cookie in $cookies) {
            if ([string]::IsNullOrEmpty($cookie)) { continue }
            $parts = $cookie -split '=', 2
            if ($parts.Count -eq 2) { $result[$parts[0].Trim()] = $parts[1].Trim() }
        }
        return $result
    }

    function GetSessionCookie ($Req) {
        $cookieHeader = $Req.Headers["Cookie"]
        if (-not [string]::IsNullOrEmpty($cookieHeader)) {
            $cookies = ParseCookies $cookieHeader
            if ($cookies.ContainsKey("LAPS_SESSION")) { return $cookies["LAPS_SESSION"] }
        }
        return $null
    }

    function LogActivity ($ActivityType, $Username, $IPAddress, $Details, $Status = "SUCCESS", $Hostname = "") {
        try {
            [System.Threading.Monitor]::Enter($SyncState.LogLock)
            try {
                $timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
                $dateForFile = Get-Date -Format "yyyy-MM-dd"
                
                $activityLogFile = Join-Path $SyncState.LogPath "LAPS_Activity_$dateForFile.log"
                $logEntry = @{
                    Timestamp    = $timestamp
                    ActivityType = $ActivityType
                    Username     = $Username
                    IPAddress    = $IPAddress
                    Hostname     = $Hostname
                    Details      = $Details
                    Status       = $Status
                } | ConvertTo-Json -Compress
                
                Add-Content -Path $activityLogFile -Value $logEntry -Encoding UTF8 -ErrorAction SilentlyContinue
                
                if ($ActivityType -in @("LOGIN", "LOGOUT", "LOGIN_FAILED", "LOGIN_BLOCKED")) {
                    $statusLogFile = Join-Path $SyncState.PortalStatusPath "Portal_Status_$dateForFile.log"
                    $statusEntry = "[$timestamp] [$ActivityType] User: $Username | IP: $IPAddress | Status: $Status"
                    if ($ActivityType -eq "LOGIN") { $statusEntry += " | EVENT: Portal Opened and Active" }
                    elseif ($ActivityType -eq "LOGOUT") { $statusEntry += " | EVENT: Portal Closed/Logged Out" }
                    Add-Content -Path $statusLogFile -Value $statusEntry -Encoding UTF8 -ErrorAction SilentlyContinue
                }
                Write-Host "[ACTIVITY] [$timestamp] [$ActivityType] User: $Username | IP: $IPAddress | Status: $Status | Details: $Details" -ForegroundColor Cyan
            } finally {
                [System.Threading.Monitor]::Exit($SyncState.LogLock)
            }
        } catch {}
    }

    function SaveAuthorizedUsersToFile {
        try {
            $userFile = Join-Path $SyncState.UserDetailsPath "authorized_users.json"
            [System.Threading.Monitor]::Enter($SyncState.AuthorizedUsers)
            try {
                $json = $SyncState.AuthorizedUsers | ConvertTo-Json -ErrorAction SilentlyContinue
                Set-Content -Path $userFile -Value $json -Encoding UTF8 -Force -ErrorAction SilentlyContinue
            } finally {
                [System.Threading.Monitor]::Exit($SyncState.AuthorizedUsers)
            }
        } catch {}
    }

    function CreateSession ($Username, $UserFullName, $IsAdmin, $IPAddress) {
        try {
            $sessionId = [System.Convert]::ToBase64String((1..32 | ForEach-Object { [byte](Get-Random -Minimum 0 -Maximum 256) })) -replace '\+', '-' -replace '/', '_' -replace '=', ''
            [System.Threading.Monitor]::Enter($SyncState.SessionLock)
            try {
                $SyncState.SessionData[$sessionId] = @{
                    User         = $Username
                    UserName     = $UserFullName
                    IsAdmin      = $IsAdmin
                    LoginTime    = Get-Date
                    LastActivity = Get-Date
                    IPAddress    = $IPAddress
                }
            } finally {
                [System.Threading.Monitor]::Exit($SyncState.SessionLock)
            }
            return $sessionId
        } catch { return $null }
    }

    function IsSessionValid ($SessionId) {
        if ([string]::IsNullOrEmpty($SessionId)) { return $false }
        [System.Threading.Monitor]::Enter($SyncState.SessionLock)
        try {
            if (-not $SyncState.SessionData.ContainsKey($SessionId)) { return $false }
            $session = $SyncState.SessionData[$SessionId]
            $elapsed = (Get-Date) - $session.LastActivity
            if ($elapsed.TotalMinutes -ge $SyncState.SESSION_TIMEOUT_MINUTES) {
                $SyncState.SessionData.Remove($SessionId)
                return $false
            }
            $session.LastActivity = Get-Date
            return $true
        } finally {
            [System.Threading.Monitor]::Exit($SyncState.SessionLock)
        }
    }

    function GetSessionUser ($SessionId) {
        if ([string]::IsNullOrEmpty($SessionId)) { return "" }
        [System.Threading.Monitor]::Enter($SyncState.SessionLock)
        try {
            if ($SyncState.SessionData.ContainsKey($SessionId)) { return $SyncState.SessionData[$SessionId].User }
        } finally { [System.Threading.Monitor]::Exit($SyncState.SessionLock) }
        return ""
    }

    function GetSessionUserName ($SessionId) {
        if ([string]::IsNullOrEmpty($SessionId)) { return "" }
        [System.Threading.Monitor]::Enter($SyncState.SessionLock)
        try {
            if ($SyncState.SessionData.ContainsKey($SessionId)) { return $SyncState.SessionData[$SessionId].UserName }
        } finally { [System.Threading.Monitor]::Exit($SyncState.SessionLock) }
        return ""
    }

    function IsSessionAdmin ($SessionId) {
        if ([string]::IsNullOrEmpty($SessionId)) { return $false }
        [System.Threading.Monitor]::Enter($SyncState.SessionLock)
        try {
            if ($SyncState.SessionData.ContainsKey($SessionId)) { return $SyncState.SessionData[$SessionId].IsAdmin }
        } finally { [System.Threading.Monitor]::Exit($SyncState.SessionLock) }
        return $false
    }

    function IsIPLocked ($IPAddress) {
        [System.Threading.Monitor]::Enter($SyncState.LoginLock)
        try {
            if ($SyncState.LoginAttempts.ContainsKey($IPAddress)) {
                $attempt = $SyncState.LoginAttempts[$IPAddress]
                $timeSinceLastAttempt = (Get-Date) - $attempt.LastAttempt
                if ($timeSinceLastAttempt.TotalMinutes -gt $SyncState.LOCKOUT_MINUTES) {
                    $SyncState.LoginAttempts.Remove($IPAddress)
                    return $false
                }
                return $attempt.FailureCount -ge $SyncState.MAX_LOGIN_ATTEMPTS
            }
            return $false
        } finally { [System.Threading.Monitor]::Exit($SyncState.LoginLock) }
    }

    function RecordFailedLogin ($IPAddress) {
        [System.Threading.Monitor]::Enter($SyncState.LoginLock)
        try {
            if ($SyncState.LoginAttempts.ContainsKey($IPAddress)) {
                $attempt = $SyncState.LoginAttempts[$IPAddress]
                $attempt.FailureCount++
                $attempt.LastAttempt = Get-Date
            } else {
                $SyncState.LoginAttempts[$IPAddress] = @{
                    IPAddress    = $IPAddress
                    FailureCount = 1
                    LastAttempt  = Get-Date
                }
            }
        } finally { [System.Threading.Monitor]::Exit($SyncState.LoginLock) }
    }

    function ClearFailedLogins ($IPAddress) {
        [System.Threading.Monitor]::Enter($SyncState.LoginLock)
        try {
            if ($SyncState.LoginAttempts.ContainsKey($IPAddress)) {
                $SyncState.LoginAttempts.Remove($IPAddress)
            }
        } finally { [System.Threading.Monitor]::Exit($SyncState.LoginLock) }
    }

    function IsUserAuthorized ($Username) {
        [System.Threading.Monitor]::Enter($SyncState.AuthorizedUsers)
        try { return $SyncState.AuthorizedUsers.ContainsKey($Username.ToUpper()) }
        finally { [System.Threading.Monitor]::Exit($SyncState.AuthorizedUsers) }
    }

    function AuthenticateWithDomain ($Username, $Password) {
        try {
            $entry = New-Object System.DirectoryServices.DirectoryEntry("LDAP://IDPBGTN.COM", "IDPBGTN\$Username", $Password)
            $null = $entry.NativeObject
            $entry.Dispose()
            return $true
        } catch { return $false }
    }

    function SendHtml ($Resp, $Html, $SessionId) {
        try {
            $buffer = [System.Text.Encoding]::UTF8.GetBytes($Html)
            $Resp.ContentType = "text/html; charset=utf-8"
            $Resp.ContentLength64 = $buffer.Length
            if (-not [string]::IsNullOrEmpty($SessionId)) {
                $cookie = New-Object System.Net.Cookie("LAPS_SESSION", $SessionId)
                $cookie.Path = "/"
                $cookie.HttpOnly = $true
                $cookie.Secure = $false
                $Resp.SetCookie($cookie)
            }
            $Resp.OutputStream.Write($buffer, 0, $buffer.Length)
            $Resp.OutputStream.Close()
        } catch { try { $Resp.Close() } catch {} }
    }

    function SendJson ($Resp, $Json) {
        try {
            $buffer = [System.Text.Encoding]::UTF8.GetBytes($Json)
            $Resp.ContentType = "application/json; charset=utf-8"
            $Resp.ContentLength64 = $buffer.Length
            $Resp.OutputStream.Write($buffer, 0, $buffer.Length)
            $Resp.OutputStream.Close()
        } catch { try { $Resp.Close() } catch {} }
    }

}

# Graceful shutdown handling
$null = Register-ObjectEvent -InputObject ([System.Console]) -EventName "CancelKeyPress" -Action {
    Write-Host "`n[SHUTDOWN] Stopping server..." -ForegroundColor Yellow
    try {
        $listener.Stop()
        $listener.Close()
        $RunspacePool.Close()
        $RunspacePool.Dispose()
    } catch { }
    Write-Host "[SHUTDOWN] Server stopped successfully" -ForegroundColor Cyan
    exit 0
}

$timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
Add-Content -Path (Join-Path $PortalStatusPath "Portal_Status_$(Get-Date -Format 'yyyy-MM-dd').log") -Value "[$timestamp] [SERVER_START] Multi-threaded Portal service started on $ServerUrl" -Encoding UTF8 -ErrorAction SilentlyContinue

Write-Host "`n========================================================" -ForegroundColor Cyan
Write-Host "         LAPS Web Portal (Multi-Threaded Enabled)       " -ForegroundColor Green
Write-Host "========================================================" -ForegroundColor Cyan
Write-Host "URL:                http://$ListenIP`:$Port/" -ForegroundColor White
Write-Host "Max Concurrent:     $MaxThreads Concurrent Requests" -ForegroundColor Magenta
Write-Host "========================================================" -ForegroundColor Cyan
Write-Host "`nPortal is live! Check your browser." -ForegroundColor Green

Start-Process "chrome.exe" -ArgumentList "$ServerUrl" -ErrorAction SilentlyContinue

# Main Loop: Instantly receives requests and offloads to RunspacePool
while ($listener.IsListening) {
    try {
        $context = $listener.GetContext()

        # Clean finished tasks to avoid memory buildup
        for ($i = $RunningTasks.Count - 1; $i -ge 0; $i--) {
            if ($RunningTasks[$i].PowerShell.InvocationStateInfo.State -ne [System.Management.Automation.PSInvocationState]::Running) {
                $RunningTasks[$i].PowerShell.EndInvoke($RunningTasks[$i].AsyncResult)
                $RunningTasks[$i].PowerShell.Dispose()
                $RunningTasks.RemoveAt($i)
            }
        }

        # Dispatch handling to background thread pool instantly
        $ps = [powershell]::Create()
        $ps.RunspacePool = $RunspacePool
        $null = $ps.AddScript($WorkerScriptBlock).AddArgument($context).AddArgument($SyncState)
        $asyncResult = $ps.BeginInvoke()

        $null = $RunningTasks.Add(@{
            PowerShell  = $ps
            AsyncResult = $asyncResult
        })
    } catch [System.Net.HttpListenerException] {
        break
    } catch {
        Write-Host "[SERVER_ERROR] $_" -ForegroundColor Red
    }
}
